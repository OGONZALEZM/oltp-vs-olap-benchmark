-- C  DuckDB star, threads=3
-- Average tip by payment type and company, one quarter
-- DuckDB EXPLAIN ANALYZE: operator timings include profiling overhead; not benchmark results.
-- Q3: average tip by payment type and company, for one quarter.
SELECT p.payment_type,
       c.company_name AS company,
       count(*)       AS trips,
       avg(f.tips)    AS avg_tip
FROM fact_trips f
JOIN dim_date d    ON d.date_key = f.date_key
JOIN dim_payment p ON p.payment_key = f.payment_key
JOIN dim_company c ON c.company_key = f.company_key
WHERE d.year = 2025 AND d.quarter = 2
GROUP BY p.payment_type, c.company_name
ORDER BY 1, 2;
